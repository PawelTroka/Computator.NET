cmake_minimum_required(VERSION 3.24)
project(ComputatorGslProbe LANGUAGES C)
find_package(GSL 2.8 REQUIRED)
add_executable(gsl-probe probe.c)
target_link_libraries(gsl-probe PRIVATE GSL::gsl GSL::gslcblas)
if(UNIX)
  target_link_libraries(gsl-probe PRIVATE m)
endif()
enable_testing()
add_test(NAME gsl-values-and-layout COMMAND gsl-probe)
